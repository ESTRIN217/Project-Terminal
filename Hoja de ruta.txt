📋 Plan de Implementación Corregido (Docker Registry V2)
Obtener Token de Autenticación: Realizar una petición GET a auth.docker.io especificando el servicio (registry.docker.io) y el alcance (repository:library/debian:pull).
Descargar el Manifest: Con el token, consultar a registry-1.docker.io usando el Header Accept correcto para obtener las capas.
Descargar los Blobs en Streaming: Iterar recursiva o secuencialmente sobre cada capa usando un método de descarga genérico que evite duplicar la lógica de buffers.

💻 Código en Kotlin para la Aplicación de Terminal
Este código realiza el flujo completo: se autentica de forma anónima, busca el contenido de la imagen de Debian y descarga la capa del sistema de archivos.

import io.ktor.client.*
import io.ktor.client.call.*
import io.ktor.client.engine.cio.*
import io.ktor.client.plugins.contentnegotiation.*
import io.ktor.client.request.*
import io.ktor.client.statement.*
import io.ktor.serialization.kotlinx.json.*
import io.ktor.utils.io.*
import io.ktor.utils.io.core.*
import kotlinx.coroutines.runBlocking
import kotlinx.serialization.Serializable
import kotlinx.serialization.json.Json
import java.io.File

// Modelos de datos para el API de Docker
@Serializable data class TokenResponse(val token: String)
@Serializable data class DockerLayer(val digest: String, val size: Long)
@Serializable data class DockerManifest(val schemaVersion: Int, val layers: List<DockerLayer>)

// Endpoints globales de la infraestructura de Docker Hub
private const val AUTH_URL = "https://auth.docker.io/token?service=registry.docker.io&scope=repository:library/debian:pull"
private const val REGISTRY_BASE_URL = "https://registry-1.docker.io/v2"

fun main() = runBlocking {
    val client = HttpClient(CIO) {
        install(ContentNegotiation) {
            json(Json { ignoreUnknownKeys = true })
        }
    }

    val repository = "library/debian"
    // Usamos la etiqueta "latest" o tu hash específico para apuntar al manifiesto correcto
    val tagOrDigest = "latest" 

    // 1. Obtener Token Autenticación
    println("1. 🔑 Solicitando token de acceso público a Docker Hub...")
    val tokenResponse: TokenResponse = client.get(AUTH_URL).body()
    val token = tokenResponse.token

    // 2. Obtener el Manifest con las capas
    println("2. 📄 Obteniendo las capas (layers) de la imagen...")
    val manifestUrl = "$REGISTRY_BASE_URL/$repository/manifests/$tagOrDigest"
    
    val manifestHttpResponse = client.get(manifestUrl) {
        header("Authorization", "Bearer $token")
        header("Accept", "application/vnd.docker.distribution.manifest.v2+json")
    }
    
    if (manifestHttpResponse.status.value != 200) {
        println("❌ Error al obtener el manifiesto: ${manifestHttpResponse.status}")
        client.close()
        return@runBlocking
    }

    val manifest: DockerManifest = manifestHttpResponse.body()
    println("3. 🗜️ Se encontraron ${manifest.layers.size} capa(s) para descargar.")
    
    // 3. Descargar capas aplicando DRY (reutilizando la función de streaming)
    manifest.layers.forEachIndexed { index, layer ->
        val blobUrl = "$REGISTRY_BASE_URL/$repository/blobs/${layer.digest}"
        val outputFile = File("debian-layer-$index.tar.gz")
        
        println("Descargando capa [${index + 1}/${manifest.layers.size}]: ${layer.digest}")
        
        downloadBlobWithStreaming(client, blobUrl, token, outputFile)
    }
    
    println("\n🎉 ¡Descarga completada! Tienes el sistema de archivos raíz listo.")
    client.close()
}

/**
 * Función utilitaria y reutilizable para descargar flujos de datos grandes (Blobs).
 * Mantiene el código DRY al abstraer el manejo del ByteReadChannel.
 */
private suspend fun downloadBlobWithStreaming(client: HttpClient, url: String, token: String, targetFile: File) {
    client.prepareGet(url) {
        header("Authorization", "Bearer $token")
    }.execute { response ->
        if (response.status.value != 200) {
            println("❌ Error al descargar capa: ${response.status}")
            return@execute
        }

        val channel: ByteReadChannel = response.bodyAsChannel()
        targetFile.outputStream().use { output ->
            while (!channel.isClosedForRead) {
                val packet = channel.readRemaining(DEFAULT_BUFFER_SIZE.toLong())
                while (!packet.isEmpty) {
                    val bytes = packet.readBytes()
                    output.write(bytes)
                }
            }
        }
        println("✅ Capa guardada con éxito en: ${targetFile.absolutePath}")
    }
}
